cmake_minimum_required(VERSION 3.25.1)
project(TauPicoGKReadback LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    message(FATAL_ERROR "The desktop PicoGK native payload currently supports Darwin arm64 only")
endif()
add_library(tau-picogk-readback SHARED mesh-readback.cpp)
target_compile_features(tau-picogk-readback PRIVATE cxx_std_20)
target_include_directories(tau-picogk-readback PRIVATE "${PICOGK_SOURCE}/API")
target_link_libraries(tau-picogk-readback PRIVATE "${PICOGK_LIBRARY}")
target_compile_definitions(tau-picogk-readback PRIVATE PICOGK_BUILD_LIBRARY)
set_target_properties(tau-picogk-readback PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path")
