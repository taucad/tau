cmake_minimum_required(VERSION 3.25.1)
project(TauPicoGKNative LANGUAGES C CXX)
if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(TAU_PICOGK_DARWIN ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(TAU_PICOGK_DARWIN OFF)
else()
    message(FATAL_ERROR "The PicoGK native payload supports Darwin arm64 and Linux x86_64 (glibc) only")
endif()
if(NOT PICOGK_SOURCE OR NOT TAU_NATIVE_DEPS)
    message(FATAL_ERROR "PICOGK_SOURCE and TAU_NATIVE_DEPS must name verified pinned source/dependency roots")
endif()
# Keep the upstream VDB format and delayed-loading behavior. Every non-system
# dependency below is built from the manifest; never discover a Homebrew library.
set(USE_CCACHE OFF CACHE BOOL "" FORCE)
set(USE_PKGCONFIG OFF CACHE BOOL "" FORCE)
set(CONCURRENT_MALLOC "None" CACHE STRING "" FORCE)
set(CMAKE_PREFIX_PATH "${TAU_NATIVE_DEPS}")
set(OPENVDB_CORE_STATIC ON CACHE BOOL "" FORCE)
set(OPENVDB_CORE_SHARED OFF CACHE BOOL "" FORCE)
set(OPENVDB_BUILD_BINARIES OFF CACHE BOOL "" FORCE)
set(OPENVDB_BUILD_UNITTESTS OFF CACHE BOOL "" FORCE)
set(OPENVDB_BUILD_PYTHON_MODULE OFF CACHE BOOL "" FORCE)
set(OPENVDB_BUILD_NANOVDB OFF CACHE BOOL "" FORCE)
set(OPENVDB_USE_DELAYED_LOADING ON CACHE BOOL "" FORCE)
set(USE_BLOSC ON CACHE BOOL "" FORCE)
set(USE_ZLIB ON CACHE BOOL "" FORCE)
set(USE_IMATH_HALF OFF CACHE BOOL "" FORCE)
set(USE_STATIC_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(BLOSC_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(Blosc_INCLUDE_DIR "${TAU_NATIVE_DEPS}/include" CACHE PATH "" FORCE)
set(Blosc_LIBRARY_RELEASE "${TAU_NATIVE_DEPS}/lib/libblosc.a" CACHE FILEPATH "" FORCE)
set(Tbb_INCLUDE_DIR "${TAU_NATIVE_DEPS}/include" CACHE PATH "" FORCE)
if(TAU_PICOGK_DARWIN)
    set(Tbb_tbb_LIBRARY_RELEASE "${TAU_NATIVE_DEPS}/lib/libtbb.12.dylib" CACHE FILEPATH "" FORCE)
else()
    set(Tbb_tbb_LIBRARY_RELEASE "${TAU_NATIVE_DEPS}/lib/libtbb.so.12" CACHE FILEPATH "" FORCE)
endif()
set(Boost_ROOT "${TAU_NATIVE_DEPS}" CACHE PATH "" FORCE)
set(Boost_DIR "${TAU_NATIVE_DEPS}/lib/cmake/Boost-1.92.0" CACHE PATH "" FORCE)
set(Boost_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(Boost_NO_SYSTEM_PATHS ON CACHE BOOL "" FORCE)
if(TAU_PICOGK_DARWIN)
    set(ZLIB_LIBRARY "${CMAKE_OSX_SYSROOT}/usr/lib/libz.tbd" CACHE FILEPATH "" FORCE)
    set(ZLIB_INCLUDE_DIR "${CMAKE_OSX_SYSROOT}/usr/include" CACHE PATH "" FORCE)
else()
    # The builder image's system zlib (libz.so.1), as Darwin links the SDK's.
    set(ZLIB_LIBRARY "/usr/lib64/libz.so" CACHE FILEPATH "" FORCE)
    set(ZLIB_INCLUDE_DIR "/usr/include" CACHE PATH "" FORCE)
    # Headless: GLFW keeps only its null platform, so the engine links no X11 or Wayland library. The viewer is
    # never opened by the hosted worker; glad and the ImGui OpenGL backend load libGL only when a viewer starts.
    set(GLFW_BUILD_X11 OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(PICOGK_UPDATE_SUBMODULES OFF CACHE BOOL "" FORCE)
add_subdirectory("${PICOGK_SOURCE}" upstream)
if(TAU_PICOGK_DARWIN)
    set_target_properties(picogk PROPERTIES VERSION 26.2 SOVERSION 26.2
        INSTALL_NAME_DIR "@loader_path" BUILD_WITH_INSTALL_NAME_DIR ON)
else()
    # `picogk.26.2.so`, the name .NET probes for PicoGK's `picogk.26.2` import, with that soname. The static
    # dependencies stay private to the engine; libtbb.so.12 is found beside it.
    set_target_properties(picogk PROPERTIES OUTPUT_NAME "picogk.26.2" SUFFIX ".so"
        BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_link_options(picogk PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:--no-undefined")
    # The ImGui GLFW backend reaches for X11 and Wayland natives on Linux; the headless GLFW has neither.
    target_compile_definitions(picogk PRIVATE IMGUI_IMPL_GLFW_DISABLE_X11 IMGUI_IMPL_GLFW_DISABLE_WAYLAND)
endif()
add_library(tau-picogk-readback SHARED mesh-readback.cpp)
target_compile_features(tau-picogk-readback PRIVATE cxx_std_20)
target_include_directories(tau-picogk-readback PRIVATE "${PICOGK_SOURCE}/API")
add_dependencies(tau-picogk-readback picogk)
if(TAU_PICOGK_DARWIN)
    target_link_libraries(tau-picogk-readback PRIVATE "$<TARGET_FILE:picogk>")
else()
    # By file name: CMake turns the engine's path into `-lpicogk.26.2`, which names a `lib`-prefixed file.
    target_link_libraries(tau-picogk-readback PRIVATE "-L$<TARGET_FILE_DIR:picogk>" "-l:$<TARGET_FILE_NAME:picogk>")
endif()
target_compile_definitions(tau-picogk-readback PRIVATE PICOGK_BUILD_LIBRARY)
if(TAU_PICOGK_DARWIN)
    set_target_properties(tau-picogk-readback PROPERTIES PREFIX ""
        BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path")
else()
    set_target_properties(tau-picogk-readback PROPERTIES PREFIX ""
        BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_link_options(tau-picogk-readback PRIVATE "LINKER:--no-undefined")
endif()
# Explicit qualification target: never compile probes into the shipped payload.
add_executable(tau-picogk-mesh-copy-test EXCLUDE_FROM_ALL tests/mesh-copy.cpp)
target_compile_features(tau-picogk-mesh-copy-test PRIVATE cxx_std_20)
target_include_directories(tau-picogk-mesh-copy-test PRIVATE "${PICOGK_SOURCE}/API")
target_link_libraries(tau-picogk-mesh-copy-test PRIVATE ${CMAKE_DL_LIBS})
